cmake_minimum_required(VERSION 3.20)
project(limitbreak LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "LimitBreak requires Windows x86 (-A Win32).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ASHITA_SDK "" CACHE PATH "Installed Ashita 4.30 SDK directory")
set(MINHOOK_SOURCE "" CACHE PATH "MinHook c3fcafdc10146beb5919319d0683e44e3c30d537")
if(NOT EXISTS "${ASHITA_SDK}/Ashita.h" OR NOT EXISTS "${MINHOOK_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Supply ASHITA_SDK and MINHOOK_SOURCE.")
endif()
add_subdirectory("${MINHOOK_SOURCE}" minhook EXCLUDE_FROM_ALL)
set(CAPSTONE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/deps/capstone-5.0.9/src" CACHE PATH "Capstone 5.0.9 source")
if(NOT EXISTS "${CAPSTONE_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Supply CAPSTONE_SOURCE (5.0.9); see docs/NATIVE_DISCOVERY.md.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
set(BUILD_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(CAPSTONE_ARCHITECTURE_DEFAULT OFF CACHE BOOL "" FORCE)
set(CAPSTONE_X86_SUPPORT ON CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CAPSTONE_BUILD_CSTOOL OFF CACHE BOOL "" FORCE)
set(CAPSTONE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CAPSTONE_SOURCE}" capstone EXCLUDE_FROM_ALL)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(NATIVE_RULES "${CMAKE_CURRENT_BINARY_DIR}/generated/discovery_rules.generated.h")
add_custom_command(OUTPUT "${NATIVE_RULES}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_native_rules.py" "${NATIVE_RULES}"
  DEPENDS tools/generate_native_rules.py tools/discovery_rules.py VERBATIM)
add_library(discoverycore STATIC src/discovery.cpp "${NATIVE_RULES}")
target_include_directories(discoverycore PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(discoverycore PRIVATE capstone_static)
target_compile_options(discoverycore PRIVATE /W4 /WX)
add_library(poolcore STATIC src/patch.cpp)
target_include_directories(poolcore PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(poolcore PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(poolcore PUBLIC bcrypt)
target_compile_options(poolcore PRIVATE /W4 /WX)
add_library(limitbreak SHARED src/plugin.cpp src/exports.def)
target_include_directories(limitbreak SYSTEM PRIVATE "${ASHITA_SDK}")
target_link_libraries(limitbreak PRIVATE poolcore minhook discoverycore)
target_compile_options(limitbreak PRIVATE /W4 /WX)
add_executable(limitbreak_tests tests/tests.cpp)
target_include_directories(limitbreak_tests SYSTEM PRIVATE "${ASHITA_SDK}")
target_link_libraries(limitbreak_tests PRIVATE poolcore minhook)
target_compile_options(limitbreak_tests PRIVATE /W4 /WX)
enable_testing()
add_test(NAME limitbreak_native COMMAND limitbreak_tests $<TARGET_FILE:limitbreak>)
add_executable(startup_tests tests/startup_tests.cpp)
target_include_directories(startup_tests SYSTEM PRIVATE "${ASHITA_SDK}")
target_link_libraries(startup_tests PRIVATE poolcore minhook discoverycore)
target_compile_options(startup_tests PRIVATE /W4 /WX)
add_test(NAME limitbreak_startup COMMAND startup_tests)
add_executable(discovery_check tests/discovery_check.cpp)
target_link_libraries(discovery_check PRIVATE discoverycore poolcore)
target_compile_options(discovery_check PRIVATE /W4 /WX)

# A never-loaded synthetic PE supplies portable full-graph discovery test input.
enable_language(ASM_MASM)
add_library(discovery_fixture SHARED tests/discovery_fixture.asm)
set_target_properties(discovery_fixture PROPERTIES LINKER_LANGUAGE C)
target_link_libraries(discovery_fixture PRIVATE kernel32)
target_link_options(discovery_fixture PRIVATE /NOENTRY /NODEFAULTLIB)
target_compile_options(discovery_fixture PRIVATE /safeseh)
